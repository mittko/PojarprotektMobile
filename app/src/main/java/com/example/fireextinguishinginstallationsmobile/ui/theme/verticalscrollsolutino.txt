                                            textAlign = TextAlign.Center
                                        )
                                        Checkbox(
                                            checked = model.checked,
                                            onCheckedChange = {
                                                model.checked = true
                                                model.unchecked = false
                                            },
                                        )
                                        OutlinedTextField(value = model.data,
                                            modifier = Modifier

                                                .padding(5.dp)

                                            ,
                                            placeholder = {
                                                Text(text = "данни", textAlign = TextAlign.Start)
                                            }, onValueChange = {
                                                model.data = it
                                            })
                                    }
                                    Column(modifier = Modifier.weight(1f),
                                        horizontalAlignment = Alignment.CenterHorizontally) {
                                        Text(
                                            text = columns[1], textAlign = TextAlign.Center
                                        )
                                        Checkbox(
                                            checked = model.unchecked,
                                            onCheckedChange = {
                                                model.checked = false
                                                model.unchecked = true
                                            })




                                        OutlinedTextField(value="",modifier = Modifier.fillMaxWidth(1f)
                                            .padding(5.dp),
                                            placeholder = {
                                                Text(text = "", textAlign = TextAlign.Start)
                                            },onValueChange = {

                                            }, enabled = false)
                                    }
                                }


                            }

                            layoutInfo = listState.layoutInfo
                            visibleInfos = layoutInfo.visibleItemsInfo
                            totalItems = layoutInfo.totalItemsCount




                            val averageItemHeightPx =
                                if (visibleInfos.isNotEmpty()) visibleInfos.sumOf { it.size }.toFloat() / visibleInfos.size
                                else 0f

                            val totalContentHeightPx =
                                if (averageItemHeightPx > 0f) averageItemHeightPx * totalItems else 0f

                            val scrollPx =
                                if (averageItemHeightPx > 0f)
                                    listState.firstVisibleItemIndex * averageItemHeightPx + listState.firstVisibleItemScrollOffset
                                else 0f

                            val minThumbPx = with(density) { 24.dp.toPx() }
                            thumbHeightPx =
                                if (containerHeightPx > 0 && totalContentHeightPx > 0f)
                                    max(minThumbPx, containerHeightPx * (containerHeightPx.toFloat() / totalContentHeightPx))
                                else minThumbPx

                            val maxScrollPx = max(totalContentHeightPx - containerHeightPx.toFloat(), 1f)
                            val scrollFraction = (scrollPx / maxScrollPx).coerceIn(0f, 1f)
                            thumbTopPx =
                                ((containerHeightPx - thumbHeightPx) * scrollFraction)
                                    .coerceIn(0f, (containerHeightPx - thumbHeightPx).coerceAtLeast(0f))

                            Log.e("THUMB",thumbTopPx.toString())
                      //      count = model.position


                        }
                }
                    Column(modifier = Modifier.fillMaxWidth(1f)
                        .fillMaxHeight()) {

                            Box(modifier = Modifier.height(with(density) {thumbHeightPx.toDp()})
                                .fillMaxWidth()
                                .offset{IntOffset(0,thumbTopPx.roundToInt())}
                                .clip(RoundedCornerShape(12.dp))) {


                                Image(
                                    painter = painterResource(id = R.drawable.ic_launcher_foreground)
                                    , contentDescription = "vertical scroll thumb",
                                    modifier = Modifier.matchParentSize(),
                                    contentScale = ContentScale.Crop
                                )
                        }
                    }
                }

            }
    }
